provider "aws" {

    }

  required_version = "~> 1.0"

    Name = "main"
  }
}

    Name = "private"

  vpc_id      = aws_vpc.main.id

  ingress {
    description = "Allow SSH from specific IP"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["192.168.1.1/32"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}
