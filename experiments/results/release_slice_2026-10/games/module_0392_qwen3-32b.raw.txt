```hcl
provider "aws" {
  region = "us-east-1"
}

resource "aws_s3_bucket" "CreateBucket" {
  force_destroy = true
  bucket        = "hx-sec-test${random_string.random_suffix.result}"

  versioning {
    enabled = true
  }

  server_side_encryption_configuration {
    rule {
      apply_server_side_encryption_by_default {
       sse_algorithm = "aws:kms"
        kms_master_key_id = aws_kms_key.EncryptionKey.arn
      }
    }
  }

  logging {
    target_bucket = aws_s3_bucket.LoggingBucket.id
    target_prefix = "logs/"
  }

  tags = {
    Name        = "HxCloudLab"
    Environment = "Lab"
  }
}

resource "aws_s3_bucket_acl" "CreateBucketAcl" {
  bucket = aws_s3_bucket.CreateBucket.id
  acl    = "private"
}

resource "aws_s3_bucket_policy" "CreateBucketPolicy" {
  bucket = aws_s3_bucket.CreateBucket.id
  policy = <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Deny",
      "Principal": "*",
      "Action": "s3:PutObject",
      "Resource": [
        "arn:aws:s3:::${aws_s3_bucket.CreateBucket.id}",
        "arn:aws:s3:::${aws_s3_bucket.CreateBucket.id}/*"
      ],
      "Condition": {
        "StringNotLike": {
          "aws:arn": [
            "arn:aws:iam::${data.aws_caller_identity.current.account_id}:user/*",
            "arn:aws:iam::${data.aws_caller_identity.current.account_id}:role/*"
          ]
        }
      }
    }
  ]
}
EOF
}

resource "aws_s3_object" "PutObject_Flag" {
  bucket = aws_s3_bucket.CreateBucket.id
  key    = "flag.txt"
  source = "./flag.txt"
  acl    = "private"
}

resource "aws_kms_key" "EncryptionKey" {
  description = "S3 encryption key"
  deletion_window_in_days = 10
}

resource "aws_kms_key_policy" "EncryptionKeyPolicy" {
  key_id = aws_kms_key.EncryptionKey.key_id
  policy = <<POLICY
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "Enable S3 encryption",
      "Effect": "Allow",
      "Principal": {
        "Service": "s3.amazonaws.com"
      },
      "Action": [
        "kms:Encrypt",
        "kms:Decrypt",
        "kms:ReEncrypt*",
        "kms:GenerateDataKey*",
        "kms:DescribeKey"
      ],
      "Resource": "*",
      "Condition": {
        "StringEquals": {
          "kms:CallerAccount": "${data.aws_caller_identity.current.account_id}",
          "kms:ViaService": "s3.amazonaws.com"
        }
      }
    },
    {
      "Sid": "Allow key management",
      "Effect": "Allow",
      "Principal": {
        "AWS": "arn:aws:iam::${data.aws_caller_identity.current.account_id}:root"
      },
      "Action": [
        "kms:Create*",
        "kms:Describe*",
        "kms:Enable*",
        "kms:List*",
        "kms:PutKeyPolicy",
        "kms:ScheduleKeyDeletion",
        "kms:UpdateKeyDescription"
      ],
      "Resource": "*"
    }
  ]
}
POLICY
}

resource "aws_s3_bucket" "LoggingBucket" {
  bucket = "hx-sec-test-logs${random_string.random_suffix.result}"
  acl    = "private"
}

resource "random_string" "random_suffix" {
  length  = 5
  special = false
  upper   = false
}

data "aws_caller_identity" "current" {}

output "AwsCloud_BucketName" {
  description = "S3 Bucket Name"
  value       = "${aws_s3_bucket.CreateBucket.bucket}.s3.us-east-1.amazonaws.com"
}

resource "aws_s3_bucket_public_access_block" "CreateBucketBlock" {
  bucket = aws_s3_bucket.CreateBucket.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

resource "aws_s3_bucket_lifecycle_configuration" "CreateBucketLifecycle" {
  bucket = aws_s3_bucket.CreateBucket.id

  rule {
    name    = "Expire old objects"
    status  = "Enabled"
    filter {
      prefix = "logs/"
    }
    transition {
      days          = 30
      storage_class = "STANDARD_IA"
    }
    expiration {
      days = 90
    }
  }
}

resource "aws_s3_bucket_notification" "CreateBucketNotification" {
  bucket = aws_s3_bucket.CreateBucket.id

  lambda_function {
    lambda_function_arn = "arn:aws:lambda:us-east-1:${data.aws_caller_identity.current.account_id}:function:example"
    events              = ["s3:ObjectCreated:*"]
    filter_prefix       = "flag.txt"
  }
}

terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "4.10.0"
    }
  }
}
```
